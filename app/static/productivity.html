function calculateDiscount(price, customerType) {
    // Copilot suggests proper validation
    if (!price || price < 0) throw new Error('Invalid price');
    
    // Copilot provides discount logic
    const discountRates = {
        'premium': 0.20,
        'standard': 0.10,
        'basic': 0.05
    };
    
    return price * (1 - (discountRates[customerType] || 0));
}
